    [STAThread]
    private static int Main(string[] args)
    {
        Root = Path.GetFullPath(args.Length == 0 ? "." : args[0]);
        Directory.CreateDirectory(Root);
        // Initialize resources without starting the production App or its timer/registry code.
        var app = new XinDong.App();
        app.InitializeComponent();
        app.ShutdownMode = ShutdownMode.OnExplicitShutdown;
        Baseline();
        Scheduling();
        StorageAndBoard();
        RegressionScenarios();
        var summary = new
        {
            TestedAt = DateTime.Now,
            Target = typeof(MainViewModel).Assembly.Location,
            TargetSha256 = Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(typeof(MainViewModel).Assembly.Location))),
            Runtime = Environment.Version.ToString(),
            Total = Results.Count, Passed = Results.Count - Failed, Failed, Results
        };
        File.WriteAllText(Path.Combine(Root, "results.json"),
            JsonSerializer.Serialize(summary, new JsonSerializerOptions { WriteIndented = true }), Encoding.UTF8);
        Console.WriteLine($"TOTAL {Results.Count}; PASS {Results.Count - Failed}; FAIL {Failed}");
        app.Shutdown();
        return 0;
    }

    private static void Check(string id, string expected, Func<string> run)
    {
        try
        {
            var actual = run();
            var pass = actual == expected;
            if (!pass) Failed++;
            Results.Add(new { Id = id, Pass = pass, Expected = expected, Actual = actual });
            Console.WriteLine($"{(pass ? "PASS" : "FAIL")} {id}: expected={expected}; actual={actual}");
        }
        catch (Exception ex)
        {
            while (ex is TargetInvocationException && ex.InnerException != null) ex = ex.InnerException;
            Failed++;
            Results.Add(new { Id = id, Pass = false, Expected = expected, Actual = $"{ex.GetType().Name}: {ex.Message}" });
            Console.WriteLine($"FAIL {id}: {ex.GetType().Name}: {ex.Message}");
        }
    }

    private static string N(double value) => value.ToString("0.###", CultureInfo.InvariantCulture);
    private static string M(decimal value) => value.ToString("0.####", CultureInfo.InvariantCulture);
    private static string B(bool value) => value ? "true" : "false";
    private static object? Invoke(object obj, string name, params object?[] args) =>
        obj.GetType().GetMethod(name, Private)!.Invoke(obj, args);
    private static T Get<T>(object obj, string name) => (T)obj.GetType().GetField(name, Private)!.GetValue(obj)!;
    private static void Set(object obj, string name, object? value) =>
        obj.GetType().GetField(name, Private)!.SetValue(obj, value);
    private static string NewDirectory()

        var path = Path.Combine(Root, "data", (++Serial).ToString("000"));
